Persze! Itt van egy egyszerű C# Windows Forms játék, egy "Találd el a számot" játék.

Mit tud?
A gép generál egy számot 1 és 100 között.
A játékos beír egy tippet.
A program megmondja, hogy nagyobb vagy kisebb számra kell gondolni.
Ha eltalálja, gratulál és új játékot kezd.

3. Kód
A Form1.cs elejére:



using System;
using System.Windows.Forms;
A Form1 osztályba:



public partial class Form1 : Form
{
    Random rnd = new Random();
    int titkosSzam;
 
    public Form1()
    {
        InitializeComponent();
        UjJatek();
    }
 
    private void UjJatek()
    {
        titkosSzam = rnd.Next(1, 101);
        label2.Text = "Kezdődhet a játék!";
        textBox1.Clear();
    }
 
    private void button1_Click(object sender, EventArgs e)
    {
        int tipp;
 
        if (!int.TryParse(textBox1.Text, out tipp))
        {
            label2.Text = "Kérlek számot adj meg!";
            return;
        }
 
        if (tipp < titkosSzam)
        {
            label2.Text = "Nagyobbra gondoltam!";
        }
        else if (tipp > titkosSzam)
        {
            label2.Text = "Kisebbre gondoltam!";
        }
        else
        {
            MessageBox.Show("Gratulálok! Eltaláltad!");
 
            UjJatek();
        }
    }
}

esetleg egy másik játék?
Persze! Egy Windows Forms projektben az egyik legjobb kezdő játék a reakcióidő játék.

Játékmenet
Megnyomod a Start gombot.
Pár másodperc múlva a panel színe pirosról zöldre vált.
Amint zöld lesz, rá kell kattintanod.
A program kiírja, hány ezredmásodperc alatt reagáltál.
Szükséges vezérlők
Button (btnStart)
Panel (panel1)
Label (lblIdo)
Timer (timer1)
Form1.cs


using System;
using System.Diagnostics;
using System.Windows.Forms;
 
namespace ReakcioJatek
{
    public partial class Form1 : Form
    {
        Random rnd = new Random();
        Stopwatch stopper = new Stopwatch();
        int varakozas;
 
        public Form1()
        {
            InitializeComponent();
 
            panel1.BackColor = System.Drawing.Color.Red;
 
            timer1.Interval = 100;
            timer1.Tick += timer1_Tick;
 
            panel1.Click += panel1_Click;
        }
 
        private void btnStart_Click(object sender, EventArgs e)
        {
            panel1.BackColor = System.Drawing.Color.Red;
            lblIdo.Text = "Várj a zöld színre!";
 
            varakozas = rnd.Next(2000, 6000);
            timer1.Interval = varakozas;
            timer1.Start();
        }
 
        private void timer1_Tick(object sender, EventArgs e)
        {
            timer1.Stop();
 
            panel1.BackColor = System.Drawing.Color.Green;
            stopper.Restart();
        }
 
        private void panel1_Click(object sender, EventArgs e)
        {
            if (panel1.BackColor == System.Drawing.Color.Green)
            {
                stopper.Stop();
 
                lblIdo.Text = $"Reakcióidőd: {stopper.ElapsedMilliseconds} ms";
                panel1.BackColor = System.Drawing.Color.Red;
            }
        }
    }
}
